```python
import random
import string
import uuid
from faker import Faker

fake = Faker('es_CO')

ALPHANUMERIC = string.ascii_uppercase + string.digits

# Each type with its share of the seed and the number format LegalDocumentType accepts.
LEGAL_DOCUMENT_TYPES = [
    ('DNI', 0.85, lambda: ''.join(random.choices(string.digits, k=8))),
    ('FOREIGNER_ID_CARD', 0.10, lambda: ''.join(random.choices(ALPHANUMERIC, k=random.randint(9, 12)))),
    ('PASSPORT', 0.05, lambda: ''.join(random.choices(ALPHANUMERIC, k=random.randint(6, 12)))),
]

def generate_legal_document(used):
    types, weights, generators = zip(*LEGAL_DOCUMENT_TYPES)
    index = random.choices(range(len(types)), weights=weights)[0]
    while True:
        document = (types[index], generators[index]())
        if document not in used:
            used.add(document)
            return document

def generate_mock_data_sql(num_records, filename="data-init.sql"):
    used_documents = set()
    with open(filename, 'w', encoding='utf-8') as file:
        file.write("-- Script generated by Python Mock Data Generator\n")
        
        for _ in range(num_records):
            beneficiary_id = uuid.uuid4()
            legal_document_type, legal_document = generate_legal_document(used_documents)
            names = fake.first_name()
            paternal_surname = fake.last_name()
            maternal_surname = fake.last_name()
            date_of_birth = fake.date_of_birth(minimum_age=18, maximum_age=85).strftime('%Y-%m-%d')
            ubigeo = fake.numerify(text='######')
            
            sql = f"INSERT INTO beneficiaries (id, legal_document_type, legal_document, names, paternal_surname, maternal_surname, date_of_birth, ubigeo) " \
                  f"VALUES ('{beneficiary_id}', '{legal_document_type}', '{legal_document}', '{names}', '{paternal_surname}', '{maternal_surname}', '{date_of_birth}', '{ubigeo}');\n"
            
            file.write(sql)
            

if __name__ == "__main__":
    RECORDS_TO_GENERATE = 150 
    generate_mock_data_sql(RECORDS_TO_GENERATE)
```
